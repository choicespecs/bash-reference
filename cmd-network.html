Bash Reference / Networking

Networking

curl -fsSL https://example.com            # fetch (fail on HTTP errors, silent, follow redirects)
curl -o file.zip https://example.com/f.zip
curl -X POST -H "Content-Type: application/json" -d '{"a":1}' https://api.example.com
curl -I https://example.com               # headers only
curl -w '%{http_code}\n' -o /dev/null -s https://example.com   # status code only
wget -q https://example.com/file

ssh user@host                             # remote shell
ssh user@host 'df -h'                     # run one remote command
ssh -L 8080:localhost:80 user@host        # local port forward
scp file.txt user@host:/tmp/              # copy to remote
rsync -avz ./site/ user@host:/var/www/    # efficient sync over ssh
ssh-keygen -t ed25519 && ssh-copy-id user@host

ping -c 4 host
dig example.com +short                    # DNS lookup
nc -zv host 22                            # is a port open?
ss -tulpn   # Linux  /  lsof -iTCP -sTCP:LISTEN   # macOS — listening ports
