CommandPurpose

grep

grep "error" app.log            # lines containing "error"
grep -i "error" app.log         # case-insensitive
grep -v "debug" app.log         # lines NOT matching
grep -rn "TODO" src/            # recursive, with line numbers
grep -E "warn|error" app.log    # extended regex (alternation)
grep -o "[0-9]\+\.[0-9]\+" f    # print only the match
grep -c "error" app.log         # count matching lines
grep -l "foo" *.txt             # list files that match
grep -A3 -B1 "panic" app.log    # 3 lines after, 1 before

sed

sed 's/old/new/' file           # replace first match per line
sed 's/old/new/g' file          # replace all
sed -i 's/old/new/g' file       # edit in place (GNU)
sed -i '' 's/old/new/g' file    # edit in place (macOS/BSD)
sed -n '10,20p' file            # print lines 10–20
sed '/^#/d' file                # delete comment lines
sed '/^$/d' file                # delete blank lines
sed -E 's/([a-z]+)@/\1 at /' f  # capture groups

awk

awk '{print $1}' file                     # first column (whitespace-split)
awk -F, '{print $2}' data.csv             # custom separator
awk '$3 > 100' file                       # filter rows
awk '{sum += $2} END {print sum}' file    # sum a column
awk 'NR > 1' file                         # skip header line
awk '{print NR": "$0}' file               # number lines
awk -F: '{print $1, $NF}' /etc/passwd     # first and last field
awk '{count[$1]++} END {for (k in count) print count[k], k}' access.log

Classic pipelines

# Top 10 most frequent words
tr -cs '[:alpha:]' '\n' < book.txt | tr 'A-Z' 'a-z' | sort | uniq -c | sort -rn | head

# Most common lines in a log
sort app.log | uniq -c | sort -rn | head

# Pretty-print and filter JSON
curl -s https://api.github.com/repos/torvalds/linux | jq '{name, stars: .stargazers_count}'
